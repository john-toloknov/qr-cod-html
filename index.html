// ========================================== // 22. СКАНЕР ЧЕРЕЗ САЙТ (Пакетная запись оборудования и кабеля) // ========================================== function doGet(e) { try { var action = e.parameter.action || ""; var dateObj = new Date(); var timeZone = Session.getScriptTimeZone() || "Asia/Almaty"; var dateStr = Utilities.formatDate(dateObj, timeZone, "dd.MM.yyyy"); var timeFormatted = Utilities.formatDate(dateObj, timeZone, "HH:mm"); var ss = SpreadsheetApp.getActiveSpreadsheet(); var author = e.parameter.author || "Сотрудник"; var installer = e.parameter.installer || ""; // Если это пакетная отправка пачки (batchSave) if (action === "batchSave" && e.parameter.batchData) { var batchData = JSON.parse(e.parameter.batchData); if (!batchData || batchData.length === 0) { return ContentService.createTextOutput("Error: Empty batch").setMimeType(ContentService.MimeType.TEXT); } var routersRows = []; var stbRows = []; var cableRows = []; var telegramItemsList = []; // Перебираем все элементы пачки и распределяем по листам for (var i = 0; i < batchData.length; i++) { var item = batchData[i]; var mode = item.mode; // 'router', 'stb', 'cable' var serial = item.serial || ""; if (mode === "cable") { var cableName = item.cableName || "Кабель"; var meters = item.meters || ""; // Лист Кабель: [Дата, Серийный/Бухта, Название, Метраж, Монтажник, Автор] cableRows.push([dateStr, serial, cableName, meters, installer, author]); telegramItemsList.push("• Кабель: " + cableName + " — " + meters + "м (S/N: " + serial + ")"); } else if (mode === "stb") { var brandStb = "Приставка IPTV ZXV10 B860H"; // Лист Приставки (структура аналогична роутерам) var newStbRow = []; newStbRow[0] = brandStb; newStbRow[1] = ""; newStbRow[2] = serial; newStbRow[3] = ""; newStbRow[4] = ""; newStbRow[5] = ""; newStbRow[6] = dateStr; newStbRow[7] = ""; newStbRow[8] = ""; newStbRow[9] = installer; newStbRow[10] = author; stbRows.push(newStbRow); telegramItemsList.push("• Приставка: " + brandStb + " (S/N: " + serial + ")"); } else { // Роутер — задействуем автоопределение модели по серийнику var brandRouter = determineRouterModel(serial); if (!brandRouter || brandRouter === "Не определено") { brandRouter = "ROUTER MODEL ERROR"; } var newRouterRow = []; newRouterRow[0] = brandRouter; newRouterRow[1] = ""; newRouterRow[2] = serial; newRouterRow[3] = ""; newRouterRow[4] = ""; newRouterRow[5] = ""; newRouterRow[6] = dateStr; newRouterRow[7] = ""; newRouterRow[8] = ""; newRouterRow[9] = installer; newRouterRow[10] = author; routersRows.push(newRouterRow); telegramItemsList.push("• Роутер: " + brandRouter + " (S/N: " + serial + ")"); } } // Записываем пачки на соответствующие листы (пакетная вставка — быстро и эффективно) if (routersRows.length > 0) { var rSheet = ss.getSheetByName("Роутеры") || ss.insertSheet("Роутеры"); rSheet.getRange(rSheet.getLastRow() + 1, 1, routersRows.length, routersRows[0].length).setValues(routersRows); } if (stbRows.length > 0) { var sSheet = ss.getSheetByName("Приставки") || ss.insertSheet("Приставки"); sSheet.getRange(sSheet.getLastRow() + 1, 1, stbRows.length, stbRows[0].length).setValues(stbRows); } if (cableRows.length > 0) { var cSheet = ss.getSheetByName("Кабель") || ss.insertSheet("Кабель"); cSheet.getRange(cSheet.getLastRow() + 1, 1, cableRows.length, cableRows[0].length).setValues(cableRows); } // Отправка единого уведомления в Telegram со всей пачкой if (installer) { var totalCount = batchData.length; var message = "⚠️ *📦 Выдача оборудования (" + totalCount + " шт.)*\n\n" + "👤 *Редактор:* " + author + "\n" + "👷 *Монтажник:* *" + installer + "*\n\n" + "📝 *Список позиций:*\n" + telegramItemsList.join("\n") + "\n\n" + "📆 *Дата:* " + dateStr + " 🕒 *Время:* " + timeFormatted; sendTelegramMessage(DEFAULT_CHAT_ID, message); checkAndSendPersonalTelegram(installer, message); } return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT); } // Обратная совместимость для старых одиночных запросов (если вдруг где-то вызывались) var mode = e.parameter.mode || "equipment"; var type = e.parameter.type || ""; var serialNumber = e.parameter.serialNumber || ""; var installerSingle = e.parameter.installer || ""; var authorSingle = e.parameter.author || "Сотрудник"; var sheetName = (mode === "cable") ? "Кабель" : "Роутеры"; var sheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName); if (mode === "cable") { var cableNameSingle = e.parameter.cableName || ""; var metersSingle = e.parameter.meters || ""; sheet.appendRow([dateStr, serialNumber, cableNameSingle, metersSingle, installerSingle, authorSingle]); } else { var brandSingle = determineRouterModel(serialNumber) || "ROUTER MODEL ERROR"; sheet.appendRow([brandSingle, "", serialNumber, "", "", "", dateStr, "", "", installerSingle, authorSingle]); } return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT); } catch (err) { console.error("Ошибка в doGet: " + err.toString()); return ContentService.createTextOutput("Error: " + err.toString()).setMimeType(ContentService.MimeType.TEXT); } } // Вспомогательная функция для личных сообщений монтажникам function checkAndSendPersonalTelegram(installerName, message) { var targetChatId = null; for (var key in INSTALLER_CHAT_IDS) { if (installerName.toLowerCase().indexOf(key.toLowerCase()) !== -1) { targetChatId = INSTALLER_CHAT_IDS[key]; break; } } if (targetChatId && targetChatId !== DEFAULT_CHAT_ID) { sendTelegramMessage(targetChatId, message); } }
